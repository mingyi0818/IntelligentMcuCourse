第 33 章 电子密码锁（STC89C52RC）

学习目标

  • 掌握 STC89C52RC + 矩阵键盘 + LCD1602 + AT24C02 的系统集成方法
  • 学会编写矩阵键盘扫描与软件消抖算法
  • 掌握 I2C 总线协议的时序与 AT24C02 EEPROM 驱动编写
  • 理解基于有限状态机的密码输入流程设计
  • 认识密码安全的相关技术：哈希存储、防暴力破解、防物理读取

33.1 项目需求

本项目设计一款基于 STC89C52RC 的电子密码锁，功能需求如下：

  • 4 位密码：使用 4×4 矩阵键盘输入 4 位数字密码，输入显示为 `*` 防窥视。
  • 开锁与报警：密码正确时电磁锁动作开锁，蜂鸣器短鸣 1 声；密码错误 3 次蜂鸣器长鸣 10 秒报警。
  • 修改密码：长按 `*` 键 3 秒进入修改模式，需先输入旧密码验证，再输入两次新密码。
  • EEPROM 存储：密码持久化存储在 AT24C02，掉电不丢失，支持 100 万次擦写。
  • 状态显示：LCD1602 显示当前状态（待机/输入/开锁/报警）、输入位数与剩余尝试次数。

33.2 硬件设计

系统以 STC89C52RC 为核心，外围模块包括 4×4 矩阵键盘、LCD1602、有源蜂鸣器、电磁锁驱动电路、AT24C02 EEPROM。详细接线如下表：

电磁锁驱动要点：电磁锁为感性负载，断电瞬间会产生反向电动势，必须在继电器线圈两端反并联续流二极管（如 1N4007），否则可能击穿驱动三极管甚至干扰单片机。

33.3.1 矩阵键盘扫描

33.3.1.1 行列扫描法原理

4×4 矩阵键盘采用行列扫描法：依次将每一行输出低电平，其余行输出高电平，读取列状态；若有列为低电平则说明该行该列交叉处的按键被按下。配合 10~20 ms 软件消抖处理。

/* key.c - 4x4 矩阵键盘扫描 */
#include <reg52.h>
#include "key.h"

#define KEY_PORT P1

/* 按键映射表：返回 0~15 表示 16 键，0xFF 表示无键 */
code unsigned char KEY_MAP[4][4] = {
    {'1','2','3','A'},
    {'4','5','6','B'},
    {'7','8','9','C'},
    {'*','0','#','D'}
};

unsigned char Key_Scan(void) {
    unsigned char row, col, key;
    for (row = 0; row < 4; row++) {
        KEY_PORT = ~(0x01 << row);   /* 选中第 row 行，其余为 1 */
        if ((KEY_PORT & 0xF0) != 0xF0) {  /* 高 4 位有按键 */
            for (col = 0; col < 4; col++) {
                if ((KEY_PORT & (0x10 << col)) == 0) {
                    key = KEY_MAP[row][col];
                    /* 软件消抖 */
                    unsigned char i, j;
                    for (i = 0; i < 20; i++) for (j = 0; j < 100; j++);
                    while ((KEY_PORT & (0x10 << col)) == 0);  /* 等待释放 */
                    return key;
                }
            }
        }
    }
    return 0xFF;
}

33.3.2 I2C 与 AT24C02 驱动

33.3.2.1 AT24C02 存储结构与器件地址

AT24C02 是 2 Kbit（256 字节）I2C EEPROM，器件地址 0xA0（写）/0xA1（读）。STC89C52RC 没有硬件 I2C，需要用 GPIO 模拟 SCL/SDA 时序：

/* i2c.c - 软件 I2C 与 AT24C02 驱动 */
#include <reg52.h>
#include "i2c.h"

sbit SCL = P3^6;
sbit SDA = P3^7;

static void I2C_Delay(void) {
    unsigned char i = 5;
    while (i--);   /* 约 5us，11.0592MHz 下 */
}

void I2C_Start(void) {
    SDA = 1; SCL = 1; I2C_Delay();
    SDA = 0; I2C_Delay();
    SCL = 0; I2C_Delay();
}

void I2C_Stop(void) {
    SDA = 0; SCL = 1; I2C_Delay();
    SDA = 1; I2C_Delay();
}

bit I2C_WriteByte(unsigned char dat) {
    bit ack;
    for (unsigned char i = 0; i < 8; i++) {
        SDA = (dat & 0x80) ? 1 : 0;
        dat <<= 1;
        I2C_Delay();
        SCL = 1; I2C_Delay(); SCL = 0;
    }
    SDA = 1;   /* 释放 SDA 等待应答 */
    SCL = 1; I2C_Delay();
    ack = SDA;  /* 0=ACK, 1=NACK */
    SCL = 0; I2C_Delay();
    return ack;
}

unsigned char I2C_ReadByte(bit ack) {
    unsigned char dat = 0;
    SDA = 1;   /* 主机释放 SDA */
    for (unsigned char i = 0; i < 8; i++) {
        SCL = 1; I2C_Delay();
        dat <<= 1;
        if (SDA) dat |= 0x01;
        SCL = 0; I2C_Delay();
    }
    SDA = ack ? 0 : 1;  /* ACK=0, NACK=1 */
    SCL = 1; I2C_Delay(); SCL = 0; I2C_Delay();
    return dat;
}

/* AT24C02 写入 1 字节，addr 范围 0x00~0xFF */
void EEPROM_WriteByte(unsigned char addr, unsigned char dat) {
    I2C_Start();
    I2C_WriteByte(0xA0);   /* 器件地址 + 写 */
    I2C_WriteByte(addr);   /* 存储地址 */
    I2C_WriteByte(dat);    /* 数据 */
    I2C_Stop();
    /* AT24C02 内部写周期 <5ms，等待 10ms 保证完成 */
    {
        unsigned char i, j;
        for (i = 0; i < 100; i++) for (j = 0; j < 100; j++);
    }
}

unsigned char EEPROM_ReadByte(unsigned char addr) {
    unsigned char dat;
    I2C_Start();
    I2C_WriteByte(0xA0);
    I2C_WriteByte(addr);
    I2C_Start();            /* Restart */
    I2C_WriteByte(0xA1);    /* 器件地址 + 读 */
    dat = I2C_ReadByte(1);  /* NACK 表示读完 */
    I2C_Stop();
    return dat;
}

33.3.3 主程序状态机

33.3.3.1 状态机实现

/* main.c - 主程序状态机 */
#include <reg52.h>
#include "lcd1602.h"
#include "key.h"
#include "i2c.h"

#define PWD_ADDR  0x10   /* 密码存储起始地址，4 字节 */
#define MAX_TRY   3

typedef enum { ST_IDLE, ST_INPUT, ST_UNLOCK, ST_ALARM, ST_MODIFY } State;
State g_state = ST_IDLE;
unsigned char g_input[4], g_idx = 0, g_try = 0;

void Load_Password(unsigned char *pwd) {
    for (unsigned char i = 0; i < 4; i++)
        pwd[i] = EEPROM_ReadByte(PWD_ADDR + i);
}

void Save_Password(const unsigned char *pwd) {
    for (unsigned char i = 0; i < 4; i++)
        EEPROM_WriteByte(PWD_ADDR + i, pwd[i]);
}

bit Check_Password(const unsigned char *input) {
    unsigned char stored[4];
    Load_Password(stored);
    for (unsigned char i = 0; i < 4; i++)
        if (input[i] != stored[i]) return 0;
    return 1;
}

void main(void) {
    unsigned char key, default_pwd[4] = {'1','2','3','4'};

    LCD_Init();
    /* 首次上电检测：若 EEPROM 内容非法（0xFF），写入默认密码 */
    if (EEPROM_ReadByte(PWD_ADDR) == 0xFF) Save_Password(default_pwd);

    LCD_ShowString(0, 0, "Smart Lock Ready");
    LCD_ShowString(1, 0, "Press any key...");

    while (1) {
        key = Key_Scan();
        if (key == 0xFF) continue;

        switch (g_state) {
            case ST_IDLE:
                g_idx = 0; g_try = 0;
                g_state = ST_INPUT;
                LCD_Clear();
                LCD_ShowString(0, 0, "Password:       ");
                LCD_ShowString(1, 0, "                ");
                /* 不 break，继续处理本次按键 */
            case ST_INPUT:
                if (key >= '0' && key <= '9') {
                    g_input[g_idx++] = key;
                    LCD_ShowChar(1, g_idx, '*');
                    if (g_idx >= 4) {
                        if (Check_Password(g_input)) {
                            g_state = ST_UNLOCK;
                            Beep(100);              /* 蜂鸣 100ms */
                            Relay_On();
                            LCD_ShowString(0, 0, "Unlocked!       ");
                            Delay_ms(5000);
                            Relay_Off();
                            g_state = ST_IDLE;
                        } else {
                            g_try++;
                            if (g_try >= MAX_TRY) {
                                g_state = ST_ALARM;
                                Beep(10000);          /* 长鸣 10s */
                                LCD_ShowString(0, 0, "ALARM!          ");
                                Delay_ms(10000);
                                g_state = ST_IDLE;
                            } else {
                                LCD_ShowString(0, 0, "Wrong! Try:    ");
                                LCD_ShowChar(0, 13, '0' + (MAX_TRY - g_try));
                                g_idx = 0;
                            }
                        }
                    }
                }
                break;
            case ST_UNLOCK:
            case ST_ALARM:
                /* 这两个状态忽略输入，由定时控制自动返回 */
                break;
            case ST_MODIFY:
                /* 修改密码流程略，需校验旧密码 + 输入两次新密码 */
                break;
        }
    }
}

主程序逻辑清晰：状态机驱动 + 按键扫描，关键操作（密码校验、EEPROM 读写）通过函数封装。这种分层设计便于后续扩展功能。

33.4 安全性考虑

电子密码锁涉及物理安全，需从多个维度考虑防护：

  • 密码加密存储：本示例为简化教学，密码明文存于 AT24C02。生产环境应使用 SHA-256 等单向哈希存储密码摘要，比对时对输入做同样哈希再比较，避免 EEPROM 被读取后泄露密码。
  • 防暴力破解：限制尝试次数（本例 3 次），错误后锁定一段时间。可加入指数退避（第一次锁定 30s、第二次 5 分钟、第三次永久锁定需管理员解锁）。
  • 防物理读取：AT24C02 通过 I2C 总线可被外部设备读取，安全等级要求高的场景应选用带加密通信的芯片（如 ATCRYPTO160）或将密码存储在单片机内部 Flash 加密区。
  • 防侧信道攻击：密码比较应使用常数时间比较，避免因比较提前返回导致时间侧信道泄露。
  • 防重放攻击：可加入时间戳或一次性令牌机制，防止记录开锁信号后重放。

常数时间比较的 C 实现：

/* 常数时间密码比较，防止时间侧信道 */
bit Check_Password_Safe(const unsigned char *input) {
    unsigned char stored[4], diff = 0;
    Load_Password(stored);
    for (unsigned char i = 0; i < 4; i++) {
        diff |= (input[i] ^ stored[i]);   /* 异或结果累计，不提前返回 */
    }
    return (diff == 0) ? 1 : 0;
}

33.6 调试与扩展方向

常见调试问题：

  • LCD 显示乱码：检查 LCD1602 8 位数据线连接顺序是否与代码一致，对比度电位器调节至字符清晰。
  • 键盘无响应：用万用表测量行列电平；按下按键时对应行列应同时为低电平。
  • EEPROM 读写不稳定：I2C 上拉电阻 4.7kΩ 不能省；写操作后必须等待 5 ms（AT24C02 内部写周期）。
  • 电磁锁不动作：测量继电器线圈两端电压，确认 12V 供电正常；检查续流二极管极性是否反接。

扩展方向：

  • RFID 刷卡开锁：增加 RC522 模块，支持 IC 卡、手机 NFC 刷卡开锁，扩展认证方式。
  • 指纹识别：集成 AS608 指纹模块，支持指纹 + 密码双重认证。
  • 蓝牙 App：增加 HC-05 蓝牙模块，通过手机 App 远程开锁、查询开锁记录。
  • 开锁记录：利用 AT24C02 剩余空间循环存储最近 50 条开锁记录（时间、密码/卡号、成功/失败）。
  • 低功耗：待机时让 STC 进入掉电模式，键盘按键触发外部中断唤醒，电池供电可工作数月。

本章小结

  • 电子密码锁综合运用了 GPIO、I2C、矩阵键盘、LCD1602、蜂鸣器、继电器等外设知识。
  • 4×4 矩阵键盘采用行列扫描法，配合软件消抖可稳定识别按键。
  • I2C 总线协议需严格按时序图实现，AT24C02 是最常见的 EEPROM，写后需等待 5 ms。
  • 状态机是描述多状态交互逻辑的有效工具，主程序清晰易维护。
  • 安全性设计应包括密码哈希存储、防暴力破解、防物理读取、常数时间比较、防重放等多重措施。
  • 项目可扩展 RFID、指纹、蓝牙等多种认证方式，向智能家居门禁系统演进。
