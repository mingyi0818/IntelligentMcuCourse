第 22 章 FreeRTOS 基础

学习目标

  • 理解裸机程序与 RTOS 的本质区别，掌握抢占式调度的优势
  • 掌握 FreeRTOS 的任务、调度器、优先级、tick 等核心概念
  • 学会使用 STM32CubeMX 集成 FreeRTOS（基于 CMSIS-RTOS v2 接口）
  • 能够使用 osThreadNew 创建/删除任务，使用信号量、互斥量保护共享资源
  • 掌握队列通信、软件定时器、栈大小调试与优化方法

22.1 裸机与 RTOS 对比

嵌入式程序的两种基本组织形式：裸机（前后台系统）与 RTOS（实时操作系统）。前者由 main() 主循环 + 中断构成"后台+前台"；后者将系统功能拆分为多个任务，由调度器按优先级抢占式调度。

对比项裸机前后台系统FreeRTOS 多任务
主结构while(1) 主循环 + ISR多个任务并发执行
响应方式顺序执行，等待事件抢占式，高优先级立即响应
模块解耦差，所有功能挤在一个循环好，每个任务独立编码
实时性取决于主循环最坏执行时间取决于最高优先级任务响应
资源消耗极低，无需 RTOS 内核需要 5~20KB ROM、每任务 1KB 栈
学习难度低中（需理解同步原语）
适用场景简单控制、单功能多任务、需并发、通信复杂
何时引入 RTOS：当系统包含多个相互独立的并行任务（如采样+显示+通信+存储）、需要严格实时响应、或代码规模超过 5000 行时，建议使用 RTOS。

22.2 FreeRTOS 核心概念

22.2.1 任务（Task）

任务是 FreeRTOS 调度的最小单位，每个任务拥有独立的栈空间与上下文，可处于 运行/就绪/阻塞/挂起 之一。任务函数通常为 void task(void *arg) 的死循环，必须调用 vTaskDelay() 或阻塞 API 让出 CPU。

22.2.2 调度器与优先级

FreeRTOS 默认采用 抢占式优先级调度：每个 tick（默认 1ms）检查就绪任务队列，让最高优先级任务运行。相同优先级则采用 时间片轮转。STM32 使用 Cortex-M4 的 PendSV 中断实现上下文切换。

22.2.3 tick 与延时

FreeRTOS 用一个硬件定时器（通常为 SysTick）产生 1ms 周期中断，称为 tick。vTaskDelay(n) 将当前任务阻塞 n 个 tick，vTaskDelayUntil() 用于周期任务。

22.3 CubeMX 集成 FreeRTOS

  1. 在 CubeMX 的 Middleware 中勾选 FREERTOS。
  2. Interface 选择 CMSIS-RTOS v2（推荐，符合 CMSIS 标准）。
  3. Configuration：
    • TOTAL_HEAP_SIZE = 15360（15KB 堆）
    • MAX_PRIORITIES = 7
    • TICK_RATE = 1000 Hz（1ms）
    • USE_PREEMPTION = Enabled
  4. 在 Tasks and Queues 标签中可可视化创建任务、队列、信号量。
  5. 注意 SYS 时基必须改为非 SysTick（如 TIM6），避免与 FreeRTOS 冲突。

22.4 任务创建与删除

CMSIS-RTOS v2 推荐使用 osThreadNew 创建任务，参数为：任务函数、参数、属性（含优先级、栈大小、名字）。

/* main.c - FreeRTOS 任务创建示例 */
#include "cmsis_os2.h"
#include "main.h"

extern ADC_HandleTypeDef hadc1;
extern UART_HandleTypeDef huart1;

osThreadId_t adcTaskHandle;
osThreadId_t ledTaskHandle;

/* ADC 采样任务 */
void adc_task(void *argument)
{
    uint32_t raw;
    float v;
    while (1) {
        HAL_ADC_Start(&hadc1);
        HAL_ADC_PollForConversion(&hadc1, 10);
        raw = HAL_ADC_GetValue(&hadc1);
        HAL_ADC_Stop(&hadc1);
        v = raw * 3.3f / 4095.0f;
        printf("ADC=%.2fV\r\n", v);
        osDelay(500);   /* 阻塞 500ms */
    }
}

/* LED 闪烁任务 */
void led_task(void *argument)
{
    while (1) {
        HAL_GPIO_TogglePin(GPIOE, GPIO_PIN_0);
        osDelay(200);
    }
}

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_USART1_UART_Init();
    MX_ADC1_Init();

    osKernelInitialize();   /* CMSIS-RTOS v2 */

    /* 任务属性 */
    const osThreadAttr_t adcAttr = {
        .name = "adcTask",
        .stack_size = 512,
        .priority = osPriorityNormal,
    };
    const osThreadAttr_t ledAttr = {
        .name = "ledTask",
        .stack_size = 128,
        .priority = osPriorityBelowNormal,
    };

    adcTaskHandle = osThreadNew(adc_task, NULL, &adcAttr);
    ledTaskHandle = osThreadNew(led_task, NULL, &ledAttr);

    osKernelStart();   /* 启动调度器，不会返回 */
    while (1) { }
}

22.4.1 任务删除

osThreadTerminate(adcTaskHandle);  /* 删除指定任务 */
osThreadExit();                     /* 删除自身 */

22.5 信号量与互斥量

共享资源（如 USART、SPI、全局变量）在多任务下必须保护，常用方法为 信号量（Semaphore）与 互斥量（Mutex）。

同步原语用途特性
二值信号量中断与任务同步0/1，可在 ISR 中释放
计数信号量资源计数多生产者多消费者
互斥量 Mutex保护共享资源带优先级继承，不可在 ISR 用
递归互斥量同任务多次获取避免自死锁
 
/* 互斥量保护 USART 输出 */
osMutexId_t uartMutex;

void safe_printf(const char *fmt, ...)
{
    char buf[128];
    va_list ap;
    va_start(ap, fmt);
    vsnprintf(buf, sizeof(buf), fmt, ap);
    va_end(ap);

    osMutexAcquire(uartMutex, osWaitForever);
    HAL_UART_Transmit(&huart1, (uint8_t*)buf, strlen(buf), 200);
    osMutexRelease(uartMutex);
}

/* 中断与任务同步：二值信号量 */
osSemaphoreId_t adcSem;

void HAL_ADC_ConvCpltCallback(ADC_HandleTypeDef *hadc)
{
    /* 在 ISR 中释放信号量，使用 isr 后缀版本 */
    osSemaphoreRelease(adcSem);
}

void adc_task(void *arg)
{
    while (1) {
        HAL_ADC_Start_DMA(&hadc1, ...);
        osSemaphoreAcquire(adcSem, osWaitForever); /* 等待转换完成 */
        /* 处理数据 */
    }
}
关键区别：信号量可在中断中释放，无优先级继承；互斥量只能在任务中获取/释放，但有优先级继承可避免优先级反转。

22.6 队列与任务间通信

消息队列（Queue）是任务间传递结构化数据的核心机制，发送端将元素入队，接收端阻塞等待。队列本身是线程安全的，内部由 FreeRTOS 提供临界区保护。

/* 队列传递传感器数据包 */
typedef struct {
    uint16_t temp;
    uint16_t humi;
    uint16_t light;
} sensor_data_t;

osMessageQueueId_t sensorQueue;

void sensor_task(void *arg)
{
    sensor_data_t data;
    while (1) {
        data.temp  = read_temp();
        data.humi  = read_humi();
        data.light = read_light();
        osMessageQueuePut(sensorQueue, &data, 0, 0);  /* 不阻塞 */
        osDelay(1000);
    }
}

void upload_task(void *arg)
{
    sensor_data_t data;
    while (1) {
        if (osMessageQueueGet(sensorQueue, &data, NULL, osWaitForever) == osOK) {
            printf("T=%d H=%d L=%d\r\n", data.temp, data.humi, data.light);
        }
    }
}

22.7 软件定时器与事件标志组

22.7.1 软件定时器

FreeRTOS 软件定时器基于 tick 实现，无需占用硬件定时器。可设置为 单次 或 周期 两种模式。回调函数在定时器服务任务中执行，禁止阻塞。

void timer_callback(void *arg)
{
    HAL_GPIO_TogglePin(GPIOE, GPIO_PIN_1);
}

osTimerId_t ledTimer;
const osTimerAttr_t timerAttr = { .name = "ledTimer" };
ledTimer = osTimerNew(timer_callback, osTimerPeriodic, NULL, &timerAttr);
osTimerStart(ledTimer, 1000);   /* 每 1s 触发一次 */

22.7.2 事件标志组

事件标志组（Event Group）允许任务等待多个事件中的任意一个或全部，每个标志组有 24 个标志位。常用于"等待多个条件都满足"的场景。

osEventFlagsId_t evtGroup;
#define FLAG_TEMP_READY  (1 << 0)
#define FLAG_HUMI_READY  (1 << 1)

void sensor_task(void *arg)
{
    read_temp();
    osEventFlagsSet(evtGroup, FLAG_TEMP_READY);
    read_humi();
    osEventFlagsSet(evtGroup, FLAG_HUMI_READY);
}

void upload_task(void *arg)
{
    while (1) {
        /* 等待两个标志同时置位 */
        uint32_t flags = osEventFlagsWait(
            evtGroup,
            FLAG_TEMP_READY | FLAG_HUMI_READY,
            osFlagsWaitAll,
            osWaitForever);
        upload_to_cloud();
        osEventFlagsClear(evtGroup, FLAG_TEMP_READY | FLAG_HUMI_READY);
    }
}

22.8 任务栈大小调试与优化

每个任务的栈必须足够大以容纳局部变量、函数调用层级和中断嵌套。栈过大会浪费内存，过小会导致栈溢出，引发 HardFault 或难以定位的崩溃。

22.8.1 查询栈剩余空间

UBaseType_t remain = uxTaskGetStackHighWaterMark(NULL);
printf("Stack remain: %u words\r\n", (unsigned)remain);

返回值为剩余字（Word, 4 字节）数。建议每个任务运行足够时间后查询剩余空间，并将任务栈大小调整为剩余值的 1.5~2 倍。

22.8.2 启用栈溢出钩子

在 FreeRTOSConfig.h 中启用：

#define configCHECK_FOR_STACK_OVERFLOW 2

/* 在 main.c 中实现 */
void vApplicationStackOverflowHook(TaskHandle_t xTask, char *pcTaskName)
{
    printf("Stack overflow in: %s\r\n", pcTaskName);
    while (1);   /* 陷入 */
}

22.8.3 优化建议

策略说明
避免大局部数组大缓冲区改为 static 全局或堆分配
减少函数递归深度递归改为迭代或加深度限制
临界区尽量短避免在中断嵌套时栈涨
使用 heap_4.c支持内存碎片合并，适合长时间运行
启用栈钩子调试阶段必须开启

本章小结

  • 裸机程序结构简单但实时性差，RTOS 通过任务+抢占调度可实现并发与严格实时响应。
  • FreeRTOS 任务有运行/就绪/阻塞/挂起四种状态，由 tick 中断驱动调度器切换。
  • CubeMX 集成 FreeRTOS 推荐 CMSIS-RTOS v2 接口，可可视化创建任务与同步原语。
  • osThreadNew 创建任务，osDelay 让出 CPU；osThreadTerminate 删除任务。
  • 互斥量保护共享资源（带优先级继承），二值信号量用于中断-任务同步，队列用于任务间数据传递。
  • 软件定时器与事件标志组分别用于周期触发与多事件等待。
  • 使用 uxTaskGetStackHighWaterMark 调优栈大小，调试阶段必须启用栈溢出钩子。

思考题与习题

  1. 简述裸机前后台系统与 RTOS 多任务的主要区别，并说明何时应当引入 RTOS。
  2. 编写程序：使用 FreeRTOS 创建 2 个任务，分别以 200ms 和 500ms 周期闪烁 LED1 与 LED2。
  3. 互斥量与二值信号量有何区别？为什么保护共享资源必须用互斥量而非信号量？
  4. 使用消息队列实现"传感器任务采集 → 显示任务展示"的流水线，队列长度为 8。
  5. 使用软件定时器实现每 5 秒打印一次系统运行时间（HAL_GetTick）。
  6. 使用 uxTaskGetStackHighWaterMark 测量三个任务的剩余栈，并据此优化 stack_size 配置。
