第 23 章 STM32 综合实验

学习目标

  • 能够基于 FreeRTOS 设计一个完整的"环境监测节点"嵌入式系统
  • 学会从硬件平台选型、外设规划到任务划分与优先级设计的全流程工程方法
  • 掌握温湿度传感器、光敏 ADC、OLED、串口上报各模块的协同编程
  • 学会借助 AI 工具辅助调试，识别并修复常见 RTOS 崩溃问题
  • 掌握看门狗、栈溢出检测、压力测试等系统稳定性验证手段

23.1 实验目标：基于 FreeRTOS 的环境监测节点

本综合实验将前面学习的 GPIO、定时器、USART、ADC、I2C、DMA、RTC、FreeRTOS 等模块融会贯通，构建一个真实可用的 环境监测节点。该节点部署在教室、温室、机房等场景，可实时采集温湿度、光照，本地 OLED 显示并周期性通过串口上报。

功能需求：

  • 温湿度采集（DHT11/AHT20）：每 2 秒采样一次，精度 ±0.5℃ / ±3% RH
  • 光照采集（光敏电阻 + ADC1_IN0）：每 500ms 采样一次，1 LSB ≈ 0.8mV
  • OLED 显示（SSD1306，I2C）：实时刷新温湿度与光照数值
  • 串口上报（USART1，115200bps）：每 5 秒打包上报一次传感器数据
  • RTC 实时时间显示：OLED 顶栏显示年月日时分秒
  • 看门狗保护：程序卡死 4 秒后系统自动复位

23.3 任务划分与优先级设计

根据功能需求将系统拆分为 4 个任务，遵循 "实时性要求越高，优先级越高" 原则：

任务名功能周期/触发优先级栈大小
sensorTask采样温湿度与光照周期 500msosPriorityAboveNormal (5)512 word
displayTask刷新 OLED 显示队列触发osPriorityNormal (4)512 word
commTask串口打包上报周期 5sosPriorityBelowNormal (3)512 word
keyTask按键处理与界面切换信号量触发osPriorityAboveNormal (5)256 word
设计要点：sensorTask 优先级最高确保数据采集实时；displayTask 由队列触发，避免空转浪费 CPU；keyTask 使用信号量阻塞等待 EXTI 中断，无按键时不消耗 CPU。

23.4 关键代码与 AI 辅助调试示例

23.4.1 全局资源与同步原语

/* main.c - 全局资源定义 */
#include "cmsis_os2.h"

extern I2C_HandleTypeDef hi2c1;
extern ADC_HandleTypeDef hadc1;
extern UART_HandleTypeDef huart1;
extern RTC_HandleTypeDef hrtc;

/* 共享数据 */
typedef struct {
    float temperature;
    float humidity;
    uint16_t light_raw;
    float light_volt;
    uint32_t timestamp;
} sensor_data_t;

volatile sensor_data_t g_data;

/* 同步原语 */
osMutexId_t dataMutex;        /* 保护 g_data */
osSemaphoreId_t keySem;       /* 按键信号量 */
osMessageQueueId_t dispQueue; /* 显示队列 */

/* FreeRTOS 钩子 */
void vApplicationStackOverflowHook(TaskHandle_t xTask, char *pcTaskName)
{
    printf("[FATAL] Stack overflow in %s\r\n", pcTaskName);
    while (1);
}

23.4.2 传感器任务

/* AHT20 寄存器定义 */
#define AHT20_ADDR        0x70   /* 7 位 0x38 左移 1 */
#define AHT20_CMD_RESET   0xBA
#define AHT20_CMD_INIT     0xBE
#define AHT20_CMD_MEASURE  0xAC

static HAL_StatusTypeDef aht20_read(float *t, float *h)
{
    uint8_t cmd[3] = {0xAC, 0x33, 0x00};
    uint8_t buf[7];
    HAL_StatusTypeDef st;

    st = HAL_I2C_Master_Transmit(&hi2c1, AHT20_ADDR, cmd, 3, 100);
    if (st != HAL_OK) return st;
    HAL_Delay(80);   /* 测量耗时 ~80ms */
    st = HAL_I2C_Master_Receive(&hi2c1, AHT20_ADDR, buf, 7, 100);
    if (st != HAL_OK) return st;

    /* 状态字节 bit7=Busy */
    if (buf[0] & 0x80) return HAL_BUSY;

    uint32_t raw_h = ((uint32_t)(buf[1] << 12) |
                      (uint32_t)(buf[2] << 4) |
                      (buf[3] >> 4));
    uint32_t raw_t = (((uint32_t)(buf[3] & 0x0F) << 16) |
                      (uint32_t)(buf[4] << 8) |
                      buf[5]);

    *h = (float)raw_h * 100.0f / 0x100000;
    *t = (float)raw_t * 200.0f / 0x100000 - 50.0f;
    return HAL_OK;
}

void sensor_task(void *arg)
{
    float t, h;
    while (1) {
        HAL_ADC_Start(&hadc1);
        HAL_ADC_PollForConversion(&hadc1, 10);
        uint32_t raw = HAL_ADC_GetValue(&hadc1);
        HAL_ADC_Stop(&hadc1);

        if (aht20_read(&t, &h) == HAL_OK) {
            osMutexAcquire(dataMutex, osWaitForever);
            g_data.temperature = t;
            g_data.humidity    = h;
            g_data.light_raw   = raw;
            g_data.light_volt  = raw * 3.3f / 4095.0f;
            g_data.timestamp   = HAL_GetTick();
            osMutexRelease(dataMutex);

            /* 通知显示任务刷新 */
            osMessageQueuePut(dispQueue, &g_data, 0, 0);
        }
        osDelay(500);
    }
}

23.4.3 显示任务

extern void ssd1306_init(void);
extern void ssd1306_show_string(uint8_t x, uint8_t page, const char *s);

void display_task(void *arg)
{
    sensor_data_t data;
    char line[24];

    ssd1306_init();
    ssd1306_show_string(0, 0, "Env Monitor v1.0");

    while (1) {
        if (osMessageQueueGet(dispQueue, &data, NULL, osWaitForever) == osOK) {
            snprintf(line, sizeof(line), "T:%5.1f C", data.temperature);
            ssd1306_show_string(0, 2, line);
            snprintf(line, sizeof(line), "H:%5.1f %%", data.humidity);
            ssd1306_show_string(0, 3, line);
            snprintf(line, sizeof(line), "L:%4u  %.2fV",
                     data.light_raw, data.light_volt);
            ssd1306_show_string(0, 4, line);
        }
    }
}

23.4.4 通信任务

void comm_task(void *arg)
{
    char buf[96];
    RTC_TimeTypeDef sTime;
    RTC_DateTypeDef sDate;

    while (1) {
        osDelay(5000);   /* 5 秒上报一次 */

        osMutexAcquire(dataMutex, osWaitForever);
        sensor_data_t d = g_data;   /* 拷贝后立即释放锁 */
        osMutexRelease(dataMutex);

        HAL_RTC_GetDate(&hrtc, &sDate, RTC_FORMAT_BCD);
        HAL_RTC_GetTime(&hrtc, &sTime, RTC_FORMAT_BCD);

        snprintf(buf, sizeof(buf),
            "20%02X-%02X-%02X %02X:%02X:%02X,T=%.2f,H=%.2f,L=%u,V=%.3f\r\n",
            sDate.Year, sDate.Month, sDate.Date,
            sTime.Hours, sTime.Minutes, sTime.Seconds,
            d.temperature, d.humidity, d.light_raw, d.light_volt);

        HAL_UART_Transmit(&huart1, (uint8_t*)buf, strlen(buf), 200);
    }
}

23.4.5 按键任务

void HAL_GPIO_EXTI_Callback(uint16_t pin)
{
    if (pin == GPIO_PIN_2) {
        /* 在 ISR 中释放信号量，使用 FromISR 版本 */
        osSemaphoreRelease(keySem);
    }
}

void key_task(void *arg)
{
    while (1) {
        osSemaphoreAcquire(keySem, osWaitForever);
        HAL_Delay(20);   /* 软件消抖 */
        if (HAL_GPIO_ReadPin(GPIOE, GPIO_PIN_2) == GPIO_PIN_RESET) {
            /* 切换显示页面：常规/调试信息 */
            static int page = 0;
            page = !page;
            printf("[KEY] switch to page %d\r\n", page);
        }
    }
}

23.4.6 main 函数

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_DMA_Init();
    MX_I2C1_Init();
    MX_ADC1_Init();
    MX_USART1_UART_Init();
    MX_RTC_Init();

    /* 看门狗：4 秒超时 */
    IWDG_HandleTypeDef hiwdg;
    hiwdg.Instance = IWDG;
    hiwdg.Init.Prescaler = IWDG_PRESCALER_64;
    hiwdg.Init.Reload = 0x0FFF;
    hiwdg.Init.Window = 0x0FFF;
    HAL_IWDG_Init(&hiwdg);

    osKernelInitialize();

    /* 创建同步原语 */
    dataMutex = osMutexNew(NULL);
    keySem    = osSemaphoreNew(1, 0, NULL);
    dispQueue = osMessageQueueNew(4, sizeof(sensor_data_t), NULL);

    osThreadNew(sensor_task,  NULL, NULL);
    osThreadNew(display_task, NULL, NULL);
    osThreadNew(comm_task,    NULL, NULL);
    osThreadNew(key_task,     NULL, NULL);

    osKernelStart();
    while (1) { }
}

23.4.7 AI 辅助调试示例

实验过程中常见的一类问题：系统运行 30~60 秒后死机，串口停止输出，OLED 卡死。借助 AI 工具（如 TRAE、Copilot）逐步定位：

  1. 症状：跑了约 1 分钟后死机，HAL_GetTick 不再增加。
  2. AI 分析假设：可能是栈溢出、看门狗超时、共享数据竞争、I2C 死锁。
  3. 插入插桩：在 vApplicationStackOverflowHook 中打印任务名，在每个任务开头打印栈剩余。
  4. 复现：发现 displayTask 栈剩余仅 12 word，远低于安全值 64。
  5. 根因：snprintf + 局部 char line[24] + I2C 调用嵌套较深，512 word 栈不够。
  6. 修复：将 displayTask 栈增至 1024 word，并在 FreeRTOSConfig.h 启用 configCHECK_FOR_STACK_OVERFLOW = 2。
  7. 验证：跑 24 小时无死机，栈剩余稳定在 200 word 以上。

23.5 系统稳定性测试与优化

23.5.1 看门狗保护

独立看门狗（IWDG）由 LSI 时钟驱动，主时钟故障时仍可工作。看门狗超时公式：

$$T_{out} = \frac{4 \cdot 2^{PR} \cdot (RL+1)}{f_{LSI}}$$

当 $PR=4$、$RL=0xFFF$、$f_{LSI}=32\text{kHz}$ 时，超时约 $\frac{4 \times 2^4 \times 4096}{32000} = 8.19\text{s}$，可在主循环或周期任务中喂狗。

/* 在 sensor_task 中喂狗（500ms 周期） */
HAL_IWDG_Refresh(&hiwdg);

23.5.2 栈溢出检测

在 FreeRTOSConfig.h 中启用：

#define configCHECK_FOR_STACK_OVERFLOW 2  /* 方法2：栈顶哨兵 + 校验 */
void vApplicationStackOverflowHook(TaskHandle_t xTask, char *pcTaskName)
{
    printf("[FATAL] Stack overflow: %s\r\n", pcTaskName);
    NVIC_SystemReset();
}

23.5.3 压力测试方案

测试项方法预期
长时间运行连续运行 24 小时无死机、无栈溢出
温度冲击0~50℃ 反复切换AHT20 数据无突变
断电恢复随机断电 5 次VBAT 维持 RTC，复位后时间正确
串口突发突发发送 1KB 数据无丢包，DMA 接收完整
按键连击1 秒内按 20 次无信号量丢失，无任务死锁

23.5.4 优化建议

  • 降低通信任务频率：5 秒上报一次足够，避免占用 UART 过久。
  • 使用 DMA 收发 UART：CPU 占用可从 30% 降至 5%。
  • OLED 局部刷新：仅刷新变化的字符区域，整体刷新耗时可减少 60%。
  • 低功耗优化：在传感器采集间隙使用 Tickless Idle，可将整体功耗降低 30~50%。
  • 日志分级：DEBUG/INFO/ERROR 三级日志，便于生产环境关闭调试信息。

本章小结

  • 本综合实验以"环境监测节点"为载体，整合 GPIO、ADC、I2C、USART、RTC、DMA、FreeRTOS 全部知识点。
  • 硬件规划遵循"模块化、可独立测试"原则，外设引脚分配需避免冲突。
  • 任务划分应按功能模块解耦，优先级按实时性需求设置，栈大小按实际剩余空间调优。
  • 共享数据用互斥量保护，中断与任务用信号量同步，任务间数据流用消息队列传递。
  • AI 辅助调试可显著缩短问题定位时间，但需结合插桩、复现、根因分析的严谨流程。
  • 稳定性测试应覆盖长时间运行、断电恢复、温度冲击、压力负载等场景，看门狗与栈溢出钩子必须启用。

思考题与习题

  1. 本实验中 sensorTask 优先级最高，displayTask 优先级较低。如果反过来会发生什么？请分析并实验验证。
  2. 编写代码：将 OLED 显示数据由"队列触发刷新"改为"周期 1 秒刷新"，并对比两种方案的实时性与 CPU 占用。
  3. 使用 IWDG 看门狗的超时时间设置为 4 秒，编写喂狗策略并设计实验验证：删除喂狗代码后系统是否会在 4 秒后复位。
  4. 压力测试：在环境监测节点运行期间，突发性向 USART1 发送 1KB 数据，验证 DMA 接收是否完整。
  5. 讨论：将 sensorTask 与 displayTask 合并为一个任务，相比拆分方案各有何优劣？请从实时性、可维护性、CPU 占用三方面分析。
