实验10 I2C读写EEPROM

I2C (Inter-Integrated Circuit，内部集成电路总线) 是一种两线制串行通信协议，广泛用于连接低速外设。本实验使用 STC89C52RC 软件模拟 I2C 与 STM32F407 硬件 I2C 读写 AT24C02 EEPROM (256 字节)，写入数据后读回校验，掌握 I2C 协议时序与 EEPROM 读写流程。完整工程源码见 /code/stc89c52rc/10-i2c-eeprom/ 与 /code/stm32f407/10-hal-i2c-eeprom/。

一、实验目的

  • 掌握 I2C 总线协议 (START/STOP/ACK/NACK/数据传输)。
  • 掌握 STC89C52RC 软件模拟 I2C 时序的方法。
  • 掌握 STM32F407 硬件 I2C 外设的 HAL 配置。
  • 掌握 AT24C02 EEPROM 的字节写、页写、随机读操作。

二、知识点

  • I2C 总线：两线 SDA (数据) + SCL (时钟)，所有设备共用总线，通过 7 位地址区分。开漏输出需外接上拉电阻 (4.7-10 kΩ)。
  • 时序：START (SCL 高时 SDA 由高变低)、STOP (SCL 高时 SDA 由低变高)、ACK (接收方拉低 SDA 应答)。
  • 数据有效性：SCL 高电平期间 SDA 必须稳定，SCL 低电平期间才允许 SDA 变化。
  • AT24C02：256 字节 EEPROM，I2C 地址 0xA0 (写) / 0xA1 (读)，A0-A2 引脚接地。页写 (Page Write) 一次最多 8 字节，跨页需延时 5 ms。
  • 写时序：START + 设备地址(写) + ACK + 字节地址 + ACK + 数据 + ACK + STOP，之后 EEPROM 内部写周期约 5 ms。
  • 读时序：先 "伪写" 设字节地址 (START+设备地址写+字节地址)，再 START+设备地址读，连续读出数据，最后 NACK + STOP。
  • 软件模拟 vs 硬件 I2C：软件模拟用 GPIO 翻转实现时序，灵活但占用 CPU；硬件 I2C 由外设自动处理，支持 DMA，效率高。

五、代码框架

5.1 STC89C52RC 软件模拟 I2C 读写 AT24C02

工程 /code/stc89c52rc/10-i2c-eeprom/soft_i2c.c。

#include <reg52.h>
#include <intrins.h>

sbit SDA = P2 ^ 0;
sbit SCL = P2 ^ 1;

#define EEPROM_ADDR 0xA0   /* AT24C02 写地址 */

void I2C_Delay(void) {
    _nop_(); _nop_(); _nop_(); _nop_();   /* 约 4 us @11.0592MHz */
}

/* I2C 起始信号: SCL 高时 SDA 由高变低 */
void I2C_Start(void) {
    SDA = 1; SCL = 1; I2C_Delay();
    SDA = 0; I2C_Delay();   /* SDA 下降沿 */
    SCL = 0; I2C_Delay();   /* 钳住总线 */
}

/* I2C 停止信号: SCL 高时 SDA 由低变高 */
void I2C_Stop(void) {
    SDA = 0; SCL = 1; I2C_Delay();
    SDA = 1; I2C_Delay();   /* SDA 上升沿 */
}

/* 发送一个字节, 返回 ACK(0)/NACK(1) */
bit I2C_SendByte(unsigned char dat) {
    unsigned char i;
    bit ack;
    for (i = 0; i < 8; i++) {
        SDA = (dat & 0x80) ? 1 : 0;   /* 高位先发 */
        dat <<= 1;
        SCL = 1; I2C_Delay();
        SCL = 0; I2C_Delay();
    }
    /* 读 ACK */
    SDA = 1;                  /* 释放 SDA */
    SCL = 1; I2C_Delay();
    ack = SDA;                /* 0=ACK, 1=NACK */
    SCL = 0; I2C_Delay();
    return ack;
}

/* 接收一个字节, ack=0 发送 ACK, ack=1 发送 NACK */
unsigned char I2C_ReadByte(bit ack) {
    unsigned char i, dat = 0;
    SDA = 1;                  /* 释放 SDA 作输入 */
    for (i = 0; i < 8; i++) {
        SCL = 1; I2C_Delay();
        dat = (dat << 1) | SDA;
        SCL = 0; I2C_Delay();
    }
    SDA = ack;                /* 发送 ACK/NACK */
    SCL = 1; I2C_Delay();
    SCL = 0; I2C_Delay();
    SDA = 1;                  /* 释放 */
    return dat;
}

/* 向 AT24C02 写一个字节 */
void EEPROM_WriteByte(unsigned char addr, unsigned char dat) {
    I2C_Start();
    I2C_SendByte(EEPROM_ADDR);   /* 设备地址 (写) */
    I2C_SendByte(addr);          /* 字节地址 */
    I2C_SendByte(dat);           /* 数据 */
    I2C_Stop();
    /* 等待内部写周期 ~5ms */
    { unsigned int i; for (i = 0; i < 5000; i++) ; }
}

/* 从 AT24C02 读一个字节 */
unsigned char EEPROM_ReadByte(unsigned char addr) {
    unsigned char dat;
    /* 1. 伪写: 设字节地址 */
    I2C_Start();
    I2C_SendByte(EEPROM_ADDR);
    I2C_SendByte(addr);
    /* 2. 重启读 */
    I2C_Start();
    I2C_SendByte(EEPROM_ADDR | 0x01);   /* 设备地址 (读) */
    dat = I2C_ReadByte(1);              /* 读 1 字节, NACK */
    I2C_Stop();
    return dat;
}

void delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++)
        for (j = 0; j < 114; j++) ;
}

void main(void) {
    unsigned char write_data = 0x5A;
    unsigned char read_data;
    unsigned char addr = 0x10;

    EEPROM_WriteByte(addr, write_data);   /* 写入 0x5A */
    delay_ms(10);
    read_data = EEPROM_ReadByte(addr);    /* 读回 */

    /* 通过串口输出 (需先初始化) 验证 read_data == 0x5A */
    SBUF = read_data;
    while (!TI) ; TI = 0;

    while (1) { }
}

5.2 STM32F407 硬件 I2C1 (HAL 库) 读写

工程 /code/stm32f407/10-hal-i2c-eeprom/main.c。PB6=SCL, PB7=SDA。

#include "stm32f4xx_hal.h"

#define EEPROM_I2C_ADDR  (0xA0 << 1)   /* HAL 库地址需左移 1 位 */
I2C_HandleTypeDef hi2c1;

void I2C1_Init(void) {
    __HAL_RCC_I2C1_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();

    GPIO_InitTypeDef gpio = {0};
    gpio.Pin = GPIO_PIN_6 | GPIO_PIN_7;
    gpio.Mode = GPIO_MODE_AF_OD;          /* 开漏复用 */
    gpio.Pull = GPIO_PULLUP;             /* 内部上拉 */
    gpio.Speed = GPIO_SPEED_FREQ_HIGH;
    gpio.Alternate = GPIO_AF4_I2C1;       /* PB6/PB7 复用 I2C1 */
    HAL_GPIO_Init(GPIOB, &gpio);

    hi2c1.Instance = I2C1;
    hi2c1.Init.ClockSpeed = 100000;       /* 100 kHz 标准模式 */
    hi2c1.Init.DutyCycle = I2C_DUTYCYCLE_2;
    hi2c1.Init.OwnAddress1 = 0;
    hi2c1.Init.AddressingMode = I2C_ADDRESSINGMODE_7BIT;
    hi2c1.Init.DualAddressMode = I2C_DUALADDRESS_DISABLE;
    hi2c1.Init.GeneralCallMode = I2C_GENERALCALL_DISABLE;
    hi2c1.Init.NoStretchMode = I2C_NOSTRETCH_DISABLE;
    HAL_I2C_Init(&hi2c1);
}

/* 写一个字节 */
void EEPROM_WriteByte(uint8_t addr, uint8_t dat) {
    HAL_I2C_Mem_Write(&hi2c1, EEPROM_I2C_ADDR, addr,
                      I2C_MEMADD_SIZE_8BIT, &dat, 1, 100);
    HAL_Delay(5);   /* 等待内部写周期 */
}

/* 读一个字节 */
uint8_t EEPROM_ReadByte(uint8_t addr) {
    uint8_t dat;
    HAL_I2C_Mem_Read(&hi2c1, EEPROM_I2C_ADDR, addr,
                     I2C_MEMADD_SIZE_8BIT, &dat, 1, 100);
    return dat;
}

int main(void) {
    uint8_t write_data = 0x5A, read_data;

    HAL_Init();
    I2C1_Init();
    EEPROM_WriteByte(0x10, write_data);
    HAL_Delay(10);
    read_data = EEPROM_ReadByte(0x10);

    /* 验证 read_data == 0x5A, 串口输出略 */
    while (1) { }
}

六、调试要点

  • 上拉电阻：I2C 为开漏总线，SDA/SCL 必须有上拉电阻 (4.7-10 kΩ)，否则电平悬浮通信失败。模块常已集成。
  • 设备地址：AT24C02 地址 0xA0 (写) / 0xA1 (读)，A0-A2 决定地址偏移；HAL 库需左移 1 位 (7 位地址 + 读写位由硬件填)。
  • 写周期等待：每次字节写后需等待 5 ms 内部写周期，连续写会失败。页写最多 8 字节，跨页自动回卷覆盖。
  • 时序延时：软件 I2C 的 I2C_Delay 决定速率，过快 (SCL 高电平 < 4 us) 不符合标准模式 (100 kHz)，过慢影响效率。
  • 电平匹配：5V AT24C02 与 3.3V STM32 通信，若 AT24C02 在 3.3V 可工作可直连；否则需电平转换芯片。
  • 应答检查：发送字节后检查 ACK，NACK 表示无应答 (设备地址错/未连接/写周期未完成)，应重试或报错。
  • 总线死锁：异常复位时从机可能拉低 SDA，主机发 9 个 SCL 脉冲可解锁。
