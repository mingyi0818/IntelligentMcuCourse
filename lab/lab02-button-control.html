实验2 按键控制与外部中断

本实验学习 GPIO 输入与外部中断 (External Interrupt, EXTI) 的使用，掌握机械按键的硬件消抖与软件消抖方法。通过按键控制 LED 状态切换，对比查询与中断两种按键处理方式的优劣。完整工程源码见 /code/stc89c52rc/02-button-exti/ 与 /code/stm32f407/02-hal-button-exti/。

一、实验目的

  • 掌握 GPIO 输入模式配置与按键读取方法。
  • 理解机械按键抖动 (Bounce) 现象，掌握软件消抖与硬件消抖 (RC 滤波)。
  • 掌握 STC89C52RC 外部中断 INT0/INT1 的配置与中断服务函数编写。
  • 掌握 STM32F407 EXTI 外部中断的配置方法。
  • 对比查询方式与中断方式的响应速度与 CPU 占用差异。

二、知识点

  • 按键抖动：机械按键在按下与释放瞬间会产生 5-20 ms 的抖动 (Bounce)，导致电平在 0/1 之间快速跳变。必须消抖才能避免一次按下被识别为多次。
  • 软件消抖：检测到电平变化后延时 10-20 ms 再次读取，若仍为变化后状态则确认有效，否则视为抖动丢弃。
  • 硬件消抖：在按键两端并联 0.1 μF 电容，利用 RC 充放电吸收高频抖动；或使用 RC 滤波电路 ($\tau = RC$) 滤除抖动。
  • STC 外部中断：INT0 引脚 P3.2，INT1 引脚 P3.3，由 TCON 寄存器的 IT0/IT1 选择下降沿 (IT0=1) 或低电平触发 (IT0=0)，IE 寄存器的 EX0/EX1 使能中断，EA 开总中断。
  • STM32 EXTI：每个 GPIO 都可作为 EXTI 线，支持上升沿、下降沿、双沿触发，配置需经过 SYSCFG (系统配置控制器) 映射。
  • 中断优先级：STC89C52RC 固定优先级 INT0 > T0 > INT1 > T1 > 串口；STM32 使用 NVIC (Nested Vectored Interrupt Controller，嵌套向量中断控制器) 支持抢占优先级与子优先级。

四、电路连接

STC89C52RC 使用 P3.2 (INT0) 接 K1，P3.3 (INT1) 接 K2，按键另一端接地，依靠 P3 口内部上拉电阻 (约 30-50 kΩ)，按下时为低电平。LED1 接 P2.0，LED2 接 P2.1 (共阳低有效)。STM32F407 使用 PA0 接 K1 (下降沿触发)，PE0 接 LED1 (推挽高有效)。

五、代码框架

5.1 STC89C52RC 查询 + 软件消抖

查询方式轮询 P3.2 电平，检测到按下后延时消抖再确认。工程 /code/stc89c52rc/02-button-exti/poll.c。

#include <reg52.h>

sbit KEY1 = P3 ^ 2;   /* K1 接 INT0 */
sbit LED1 = P2 ^ 0;   /* LED1 共阳, 0 点亮 */

void delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++)
        for (j = 0; j < 114; j++) ;
}

/* 按键扫描: 返回 1 表示一次有效按下 */
unsigned char Key_Scan(void) {
    if (KEY1 == 0) {            /* 检测到低电平 */
        delay_ms(15);           /* 软件消抖 15 ms */
        if (KEY1 == 0) {        /* 再次确认仍为低 */
            while (!KEY1) ;     /* 等待按键释放, 避免长按重复触发 */
            return 1;           /* 返回有效按键 */
        }
    }
    return 0;
}

void main(void) {
    unsigned char led_state = 0;
    while (1) {
        if (Key_Scan()) {
            led_state = !led_state;     /* 翻转 LED 状态 */
            LED1 = led_state ? 0 : 1;   /* 共阳: 0 亮 1 灭 */
        }
    }
}

5.2 STC89C52RC 外部中断 INT0 方式

中断方式配置 INT0 下降沿触发，按键按下立即进入中断服务函数。工程 /code/stc89c52rc/02-button-exti/int0.c。

#include <reg52.h>

sbit LED1 = P2 ^ 0;   /* LED1 */

/* INT0 中断号 0, 中断服务函数 */
void INT0_ISR(void) interrupt 0 {
    unsigned int i;
    for (i = 0; i < 1500; i++) ;   /* 中断内简单消抖延时约 15 ms */
    if (INT0 == 0) {                /* 再次确认仍为低电平 */
        LED1 = !LED1;               /* 翻转 LED */
        while (!INT0) ;             /* 等待释放 */
    }
}

void main(void) {
    IT0 = 1;    /* INT0 下降沿触发 (TCON.0) */
    EX0 = 1;    /* 使能 INT0 中断 (IE.0) */
    EA  = 1;    /* 开放总中断 (IE.7) */
    LED1 = 1;   /* 初始熄灭 (共阳) */
    while (1) {
        /* 主循环空闲, CPU 可做其它工作 */
        PCON |= 0x01;   /* 进入空闲模式降低功耗, 中断可唤醒 */
    }
}

5.3 STM32F407 HAL 外部中断 PA0

使用 HAL 库配置 PA0 为 EXTI0 下降沿中断。工程 /code/stm32f407/02-hal-button-exti/main.c。

#include "stm32f4xx_hal.h"

#define LED1_PIN  GPIO_PIN_0
#define LED1_PORT GPIOE

void EXTI0_IRQHandler(void) {
    HAL_GPIO_EXTI_IRQHandler(GPIO_PIN_0);   /* 进入 HAL 统一回调 */
}

/* HAL 库外部中断回调函数 */
void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) {
    if (GPIO_Pin == GPIO_PIN_0) {
        HAL_Delay(15);   /* 软件消抖 (注意: 中断内不宜长延时, 实际用定时器更佳) */
        if (HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_0) == GPIO_PIN_RESET) {
            HAL_GPIO_TogglePin(LED1_PORT, LED1_PIN);   /* 翻转 LED */
        }
    }
}

int main(void) {
    GPIO_InitTypeDef GPIO_InitStruct = {0};

    HAL_Init();
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOE_CLK_ENABLE();

    /* 配置 PA0 为外部中断输入, 下降沿, 上拉 */
    GPIO_InitStruct.Pin  = GPIO_PIN_0;
    GPIO_InitStruct.Mode = GPIO_MODE_IT_FALLING;
    GPIO_InitStruct.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

    /* 配置 PE0 推挽输出驱动 LED */
    GPIO_InitStruct.Pin   = LED1_PIN;
    GPIO_InitStruct.Mode  = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull  = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(LED1_PORT, &GPIO_InitStruct);

    HAL_NVIC_SetPriority(EXTI0_IRQn, 2, 0);   /* 设置优先级 */
    HAL_NVIC_EnableIRQ(EXTI0_IRQn);           /* 使能 NVIC */

    while (1) { /* 主循环空闲 */ }
}

六、调试要点

  • 抖动确认：用示波器观察按键两端波形可清晰看到按下瞬间的毛刺，验证消抖必要性。
  • 消抖延时取值：通常 10-20 ms，过短除不净抖动，过长响应迟钝。本实验取 15 ms。
  • 等待释放：查询方式必须等待按键释放再返回，否则长按会连续触发；中断方式同样需在中断内等待释放。
  • 中断标志清除：STM32 进入 HAL_GPIO_EXTI_IRQHandler 会自动清除 EXTI PR 标志；STC89C52RC 外部中断标志 IE0/IE1 硬件自动清除。
  • 低电平触发陷阱：STC INT0 设为低电平触发 (IT0=0) 时，按键长按会持续触发中断，故推荐下降沿触发。
  • 中断内延时：在中断服务函数内调用 HAL_Delay 存在风险 (依赖 SysTick 中断，若优先级不够会死锁)，工程上应改用定时器或在中断内只置标志位、主循环处理。
