实验3 矩阵键盘扫描

当按键数量较多时，独立式按键会占用大量 IO 口。矩阵键盘 (Matrix Keypad) 通过行列交叉结构，用 $n+m$ 条线即可实现 $n \times m$ 个按键的识别，是嵌入式系统常用的输入方案。本实验使用 4×4 矩阵键盘，掌握行列扫描法与线反转法两种识别算法。完整工程源码见 /code/stc89c52rc/03-matrix-keypad/。

一、实验目的

  • 理解矩阵键盘的行列交叉结构，掌握节省 IO 的原理。
  • 掌握行列扫描法 (Row-Column Scanning) 的实现步骤。
  • 掌握线反转法 (Line Inversion) 的快速识别原理。
  • 能将按键值在数码管或串口输出验证。
  • 理解扫描周期与响应速度的关系。

二、知识点

  • 矩阵结构：4 行 4 列共 16 键，行线接 P1.0-P1.3，列线接 P1.4-P1.7，每个按键跨接在一条行线与一条列线之间，按键按下时该行与该列导通。
  • 行列扫描法：逐行输出低电平，其余行输出高电平，读取列线；若某列为低则该行该列交叉的按键被按下。需扫描 4 次遍历全部按键。
  • 线反转法：先将行线全输出 0、列线作输入读回，得到列号；再将列线全输出 0、行线作输入读回，得到行号；两次读取组合即可确定按键，速度更快。要求 IO 必须支持双向，STC 准双向口满足此条件。
  • 键码映射：行列值组合通过查表映射为 0-F 的键码，本实验布局为 0-9 数字键与 A-F 功能键。
  • 扫描周期：主循环每次扫描约 1-2 ms，消抖等待 10-20 ms，整体响应时间 < 50 ms 即手感良好。
  • IO 复用：4×4 = 16 键仅用 8 条 IO，相比独立按键省 8 条，是典型的 IO 复用技术。

四、电路连接

五、代码框架

5.1 行列扫描法

逐行拉低，读取列线判断按键。工程 /code/stc89c52rc/03-matrix-keypad/scan.c。

#include <reg52.h>

#define KEY_PORT P1   /* 矩阵键盘接 P1 口 */

/* 共阴数码管段码表 0-F */
unsigned char code SEG[] = {
    0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07,
    0x7F, 0x6F, 0x77, 0x7C, 0x39, 0x5E, 0x79, 0x71
};

void delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++)
        for (j = 0; j < 114; j++) ;
}

/* 行列扫描法: 返回 0-15 按键值, 0xFF 表示无键 */
unsigned char Key_Scan(void) {
    unsigned char row, col, key = 0xFF;
    for (row = 0; row < 4; row++) {
        /* 拉低当前行, 其余行高 */
        KEY_PORT = ~(0x01 << row);   /* 例如 row=0 输出 0xFE */
        if ((KEY_PORT & 0xF0) != 0xF0) {   /* 读高 4 位列线 */
            col = (KEY_PORT >> 4) & 0x0F;   /* 取列状态 */
            /* 找到为 0 的列 */
            if      (col == 0x0E) col = 0;
            else if (col == 0x0D) col = 1;
            else if (col == 0x0B) col = 2;
            else if (col == 0x07) col = 3;
            else continue;
            key = row * 4 + col;     /* 计算键码 */
            while ((KEY_PORT & 0xF0) != 0xF0) ;  /* 等待释放 */
            delay_ms(10);            /* 释放消抖 */
            return key;
        }
    }
    return 0xFF;
}

void main(void) {
    unsigned char key;
    P0 = 0x00;   /* 数码管初始熄灭 */
    while (1) {
        key = Key_Scan();
        if (key != 0xFF) {
            P0 = SEG[key];   /* 数码管显示键值 */
        }
        delay_ms(1);        /* 扫描间隔 */
    }
}

5.2 线反转法

两次读取行、列状态组合按键值，速度更快。工程 /code/stc89c52rc/03-matrix-keypad/invert.c。

#include <reg52.h>

#define KEY_PORT P1

/* 行列组合 -> 键码 查表 */
unsigned char code KEY_TABLE[] = {
    0xEE, 0xED, 0xEB, 0xE7,   /* 行0: 按键 0,1,2,3 */
    0xDE, 0xDD, 0xDB, 0xD7,   /* 行1: 按键 4,5,6,7 */
    0xBE, 0xBD, 0xBB, 0xB7,   /* 行2: 按键 8,9,A,B */
    0x7E, 0x7D, 0x7B, 0x77    /* 行3: 按键 C,D,E,F */
};

void delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++)
        for (j = 0; j < 114; j++) ;
}

unsigned char Key_Invert(void) {
    unsigned char row_read, col_read, code_val, i;

    /* 第 1 步: 行线输出 0, 列线作输入读 */
    KEY_PORT = 0x0F;            /* 行线 P1.0-P1.3 = 0, 列线 P1.4-P1.7 = 1 */
    if (KEY_PORT != 0x0F) {     /* 列线读回不全 1, 有按键 */
        delay_ms(10);            /* 消抖 */
        if (KEY_PORT != 0x0F) {
            row_read = KEY_PORT & 0x0F;    /* 读行 (实际列状态) */

            /* 第 2 步: 列线输出 0, 行线作输入读 */
            KEY_PORT = 0xF0;
            col_read = KEY_PORT & 0xF0;    /* 读列 (实际行状态) */

            code_val = col_read | row_read;  /* 组合 8 位键码 */
            while (KEY_PORT != 0xF0) ;        /* 等待释放 */
            KEY_PORT = 0x0F;
            while (KEY_PORT != 0x0F) ;

            /* 查表得键值 */
            for (i = 0; i < 16; i++) {
                if (code_val == KEY_TABLE[i]) return i;
            }
        }
    }
    return 0xFF;
}

void main(void) {
    while (1) {
        unsigned char k = Key_Invert();
        if (k != 0xFF) {
            SBUF = k;   /* 通过串口输出键值 (需先初始化串口) */
            while (!TI) ; TI = 0;
        }
    }
}

六、调试要点

  • 行列方向：实际接线需与代码中行、列定义一致，否则键码错乱。建议先用万用表蜂鸣档确认按键的行列对应关系。
  • 读取时机：写行线后立即读列线，STC 准双向口翻转需几个机器周期，必要时加 1-2 个 NOP。
  • 多键同时按下：扫描法可能误判，工程中应加保护 (如只响应单键，多键视为无效)。
  • 线反转法适用条件：IO 口必须支持双向快速切换，STC 准双向口可以；若 GPIO 为单向输入或输出则不可用。
  • 扫描频率：主循环扫描过快会浪费 CPU，过慢响应迟钝；建议 1-5 ms 一次扫描。
  • 长按处理：等待释放 (while 循环) 会阻塞扫描，若需支持长按连发，应使用状态机 + 定时器。

七、思考题

  1. 4×4 矩阵键盘用 8 条 IO 实现 16 键，相比独立按键节省了多少条 IO？若需要 64 键，矩阵方案需要多少条 IO？
  2. 行列扫描法与线反转法各有何优缺点？为什么线反转法速度更快？
  3. 线反转法对 IO 口有什么特殊要求？STM32 的 GPIO 能否直接使用线反转法？
  4. 若矩阵键盘的某条行线断路，会出现什么现象？如何通过现象定位故障？
  5. 设计一个支持 "长按连发" 的矩阵键盘扫描算法 (按住超过 0.5 s 后每 50 ms 触发一次)，描述状态机设计思路。
